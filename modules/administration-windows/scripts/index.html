Exemples pédagogiques · à tester dans la maquette

Les scripts de création et d'activation modifient AD après confirmation. Les trois autres sont des outils de contrôle/export. Aucun script n'a été exécuté sur tes VM.

activation-compte.ps1

Télécharger .ps1
powershell
# Lab ENI - saisie interactive, aucune valeur secrete en dur.
Import-Module ActiveDirectory -ErrorAction Stop
do {
    Write-Host "=== Activation du compte AD ===" -ForegroundColor Cyan
    $User = Read-Host "Identifiant"
    try {
        $Compte = Get-ADUser -Identity $User -ErrorAction Stop
        Write-Host ("Compte : " + $Compte.Name)
        if ((Read-Host "Confirmer la reinitialisation et activation (O/N)") -eq "O") {
            $Pass = Read-Host "Nouveau mot de passe" -AsSecureString
            Set-ADAccountPassword -Identity $Compte -Reset -NewPassword $Pass -ErrorAction Stop
            Enable-ADAccount -Identity $Compte -ErrorAction Stop
            Get-ADUser -Identity $Compte.DistinguishedName -ErrorAction Stop |
                Select-Object Name,SamAccountName,Enabled
        }
    }
    catch {
        Write-Host $_.Exception.Message -ForegroundColor Red
    }
    $Continuer = Read-Host "Autre utilisateur (O/N)"
} while ($Continuer -eq "O")

export-groupes.ps1

Télécharger .ps1
powershell
# Lecture seule AD ; execute depuis le repertoire d'atelier
Import-Module ActiveDirectory -ErrorAction Stop
$OU = "OU=Groupes,OU=Société ASR,DC=domED,DC=ad"
$Dossier = Join-Path (Get-Location) "livrables"
$Fichier = Join-Path $Dossier "ctrl_membres_groupes.txt"
try {
    New-Item -Path $Dossier -ItemType Directory -Force -ErrorAction Stop | Out-Null
    $Groupes = Get-ADGroup -Filter {GroupScope -eq "Global"} -SearchBase $OU -ErrorAction Stop | Sort-Object Name
    $Rapport = foreach ($Groupe in $Groupes) {
        "===================================="
        "Groupe : $($Groupe.Name)"
        "===================================="
        $Membres = @(Get-ADGroupMember -Identity $Groupe -ErrorAction Stop)
        if ($Membres.Count -eq 0) { "(Aucun membre)" }
        else { $Membres | Sort-Object Name | Select-Object -ExpandProperty Name }
        ""
    }
    if (-not $Groupes) { throw "Aucun groupe global trouve dans l'OU." }
    $Rapport | Out-File -FilePath $Fichier -Encoding UTF8 -ErrorAction Stop
    Write-Host "Export : $Fichier" -ForegroundColor Green
}
catch { Write-Host $_.Exception.Message -ForegroundColor Red }

export-utilisateur.ps1

Télécharger .ps1
powershell
# Lecture seule AD ; execute depuis le repertoire d'atelier
Import-Module ActiveDirectory -ErrorAction Stop
$User = Read-Host "Identifiant (laisser vide pour dgrenier)"
if ([string]::IsNullOrWhiteSpace($User)) { $User = "dgrenier" }
$Dossier = Join-Path (Get-Location) "livrables"
$NomFichier = if ($User -eq "dgrenier") { "detail_user_David.txt" } else { "detail_user_$($User -replace '[^a-zA-Z0-9_.-]','_').txt" }
$Fichier = Join-Path $Dossier $NomFichier
try {
    New-Item -Path $Dossier -ItemType Directory -Force -ErrorAction Stop | Out-Null
    Get-ADUser -Identity $User -Properties * -ErrorAction Stop |
        Format-List * | Out-File -FilePath $Fichier -Encoding UTF8 -Width 4096 -ErrorAction Stop
    Write-Host "Export : $Fichier" -ForegroundColor Green
}
catch { Write-Host $_.Exception.Message -ForegroundColor Red }

creation-utilisateur.ps1

Télécharger .ps1
powershell
# Exemple generique : ne remplace pas la copie des modeles du TP03.
Import-Module ActiveDirectory -ErrorAction Stop
$BaseOU = "OU=Utilisateurs,OU=Société ASR,DC=domED,DC=ad"
do {
    Write-Host "=== Creation utilisateur AD ===" -ForegroundColor Cyan
    $Prenom = (Read-Host "Prenom").Trim()
    $Nom = (Read-Host "Nom").Trim()
    if (-not $Prenom -or -not $Nom) {
        Write-Host "Prenom et nom obligatoires" -ForegroundColor Red
    }
    else {
        $Login = ($Prenom.Substring(0,1) + $Nom).ToLowerInvariant().Normalize("FormD") -replace '\p{Mn}|[^a-zA-Z0-9._-]', ''
        $Services = @("Direction","Informatique","Comptabilité")
        do {
            Write-Host "1 Direction / 2 Informatique / 3 Comptabilite"
            $Choix = Read-Host "Service"
        } until ($Choix -in "1","2","3")
        $Service = $Services[[int]$Choix - 1]
        $Params = @{
            Name = "$Prenom $Nom"
            GivenName = $Prenom
            Surname = $Nom
            DisplayName = "$Prenom $Nom"
            SamAccountName = $Login
            UserPrincipalName = "$Login@domED.ad"
            Department = $Service
            Path = "OU=$Service,$BaseOU"
            Enabled = $false
        }
        $Fonction = Read-Host "Fonction (facultatif)"
        $Telephone = Read-Host "Telephone (facultatif)"
        if ($Fonction) { $Params.Title = $Fonction }
        if ($Telephone) { $Params.OfficePhone = $Telephone }
        if ((Read-Host "Expiration au 31 decembre (O/N)") -eq "O") {
            # Expiration en debut de journee du 1er janvier suivant.
            $Params.AccountExpirationDate = (Get-Date -Year ((Get-Date).Year + 1) -Month 1 -Day 1).Date
        }
        Write-Host "$Login / $Service / compte desactive sans mot de passe"
        if ((Read-Host "Confirmer (O/N)") -eq "O") {
            try {
                if (-not $Login -or $Login.Length -gt 20) { throw "Identifiant vide ou trop long (max 20)." }
                New-ADUser @Params -ErrorAction Stop
                Write-Host "Compte cree" -ForegroundColor Green
            }
            catch { Write-Host $_.Exception.Message -ForegroundColor Red }
        }
    }
    $Continuer = Read-Host "Autre utilisateur (O/N)"
} while ($Continuer -eq "O")
